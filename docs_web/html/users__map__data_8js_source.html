106/**
107 * @brief Descarga los datos de un gas específico desde Firestore.
108 * @details
109 * 1. Consulta la colección `Tipo_Gas` para obtener umbrales y colores.
110 * 2. Consulta la colección `Mapas_Diarios` para obtener los puntos de medición.
111 * 3. Rellena `globalDatosCache` con los valores numéricos reales.
112 * 4. Genera los puntos visuales (normalizados 0-1) para el Heatmap.
113 *
114 * @param {string} fechaISO - Fecha solicitada.
115 * @param {string} tipoGas - Tipo de gas.
116 */
117async function descargarGasIndividual(fechaISO, tipoGas) {
118 const partes = fechaISO.split('-');
119 const fechaFormateada = `${partes[2]}-${partes[1]}-${partes[0]}`; // DD-MM-YYYY
120 const idMap = `${tipoGas}_${fechaFormateada}`;
121
122 try {
123 let nivelesBBDD = [];
124 let unidad = UNIDADES_ESTANDAR[tipoGas] || "ppm";
125 let configGradiente = null;
126
127 // 1. Obtener Configuración del Gas (Umbrales)
128 try {
129 const docConfig = await db.collection("Tipo_Gas").doc(tipoGas).get();
130 if (docConfig.exists) {
131 const data = docConfig.data();
132 if (data.Umbrales?.Unidad) unidad = data.Umbrales.Unidad;
133 if (data.Umbrales?.niveles) {
134 nivelesBBDD = data.Umbrales.niveles;
135 configGradiente = procesarColoresDeBBDD(nivelesBBDD);
136 }
137 }
138 } catch (e) { console.warn("Usando config default para", tipoGas); }
139
140 // 2. Obtener Datos del Mapa (Puntos)
141 const docMapa = await db.collection("Mapas_Diarios").doc(idMap).get();

143 if (!docMapa.exists) {
144 console.warn(" No hay datos para esta fecha en BBDD.");
145 globalDatosCache = [];
146 return { puntos: [], gradiente: configGradiente, unidad };
147 }
148
149 const arrayPuntos = docMapa.data().Puntos_array || [];
150
151 // --- DEBUG IMPORTANTE: Ver qué valores llegan realmente ---
152 if (arrayPuntos.length > 0) {

159 console.warn(" ALERTA: Los valores parecen normalizados (0-1). El mapa saldrá coloreado pero el dato numérico será muy bajo (0.x). Revisa si guardas el valor real en BBDD.");

184/**
185 * @brief Calcula el nivel de contaminación estimado en una coordenada específica.
186 * @details
187 * Utiliza el algoritmo de Interpolación de Ponderación de Distancia Inversa (IDW).
188 * - Busca puntos en `globalDatosCache`.
189 * - Pondera los valores cercanos (más peso cuanto más cerca).
190 * - Si está fuera del rango (`DISTANCIA_CORTE`), devuelve 0.
191 * - Incluye correcciones para evitar el "efecto borde vacío".
192 *
193 * @param {number} lat - Latitud del punto a consultar.
194 * @param {number} lng - Longitud del punto a consultar.
195 * @return {number} Valor interpolado de contaminación.
196 */
197window.calcularContaminacionEnUbicacion = function(lat, lng) {
198 if (!globalDatosCache || !globalDatosCache.length) return 0;
199
200 // Distancias en GRADOS (aprox: 0.01 grados ~ 1.1km)
201 // Aumentamos un poco el radio para asegurar que pille datos si estás cerca
202 const DISTANCIA_CORTE = 0.05; // ~5.5km (Si estás más lejos, devuelve 0)
203 const RADIO_VISUAL = 0.035; // ~3.8km (Radio de influencia para interpolar)
204
205 let sumaPonderada = 0;
206 let pesoTotal = 0;
207 let distMasCercana = Infinity;
208 let valorMasCercano = 0;
209
210 globalDatosCache.forEach(pt => {
211 // Distancia Euclidiana simple (suficiente para distancias cortas)
212 const d = Math.sqrt(Math.pow(lat - pt[0], 2) + Math.pow(lng - pt[1], 2));
213
214 // Rastrear siempre el punto más cercano absoluto
215 if (d < distMasCercana) {
216 distMasCercana = d;
217 valorMasCercano = pt[2];
218 }
219
220 // Interpolación IDW (Inverse Distance Weighting)
221 if (d < RADIO_VISUAL) {
222 // Factor de suavizado (0.001) para evitar división por cero si estás encima
223 const peso = 1 / Math.pow(d + 0.001, 2);
224 sumaPonderada += pt[2] * peso;
225 pesoTotal += peso;
226 }
227 });
228
229 console.log(` Cálculo Pin (${lat.toFixed(4)},${lng.toFixed(4)}): DistanciaMin=${distMasCercana.toFixed(4)}, ValorCercano=${valorMasCercano}`);
230
231 // CASO 1: Estás demasiado lejos de cualquier sensor
232 if (distMasCercana > DISTANCIA_CORTE) {
233 console.log("-> Fuera de rango (0.0)");
234 return 0.0;
235 }
236
237 // CASO 2: Estás en zona de influencia (Interpolar)
238 if (pesoTotal > 0) {
239 const resultado = parseFloat((sumaPonderada / pesoTotal).toFixed(2));
240 // FIX: Si el resultado es absurdamente pequeño (<0.1) pero hay datos cercanos grandes,
241 // forzamos el valor más cercano para evitar el "efecto borde vacío".
242 if (resultado < 0.1 && valorMasCercano > 1) {
243 return parseFloat(valorMasCercano.toFixed(2));
244 }
245 return resultado;
246 }
247
248 // CASO 3: Fallback (Mancha aislada sin peso suficiente)
249 return parseFloat(valorMasCercano.toFixed(2));
250};

254/**
255 * @brief Determina la intensidad visual (0-1) de un valor según los umbrales.
256 * @param {number} valor - Valor numérico del contaminante.
257 * @param {Array} niveles - Configuración de niveles de la BBDD.
258 * @return {number} Intensidad para el Heatmap (ej: 0.2, 0.6, 1.0).
259 */
260function obtenerIntensidadSegunNivel(valor, niveles) {
261 if (!niveles || niveles.length === 0) return 0.2;
262 let colorEncontrado = 'verde';
263 for (let nivel of niveles) {
264 let cumpleMin = (nivel.min === undefined) || (valor >= nivel.min);
265 let cumpleMax = (nivel.max === undefined) || (valor <= nivel.max);
266 if (cumpleMin && cumpleMax) {
267 let c = nivel.color ? nivel.color.toLowerCase() : 'verde';
268 if (c === 'naranja') c = 'amarillo';
269 if (c === 'morado') c = 'rojo';
270 colorEncontrado = c;
271 break;
272 }
273 }
274 return INTENSIDAD_COLORES[colorEncontrado] || 0.2;
275}

293/**
294 * @brief Genera un mapa compuesto de Calidad General (ICA).
295 * @details
296 * Descarga todos los gases disponibles, superpone sus intensidades y se queda
297 * con el peor valor (máximo) en cada punto geográfico.
298 * @param {string} fechaISO - Fecha solicitada.
299 * @return {Promise<Object>} Datos del mapa general.
300 */
301async function calcularMapaCalidadGeneral(fechaISO) {
302 const gases = ['O3', 'NO2', 'CO', 'SO2', 'CO2'];
303 const promesas = gases.map(gas => descargarGasIndividual(fechaISO, gas));
304 const resultados = await Promise.all(promesas);
305
306 const mapaUnificado = new Map();
307
308 resultados.forEach(res => {
309 if (!res.puntos) return;
310 res.puntos.forEach(pt => {
311 const key = `${pt[0].toFixed(5)}_${pt[1].toFixed(5)}`;
312 const intensidad = pt[2];
313 if (mapaUnificado.has(key)) {
314 if (intensidad > mapaUnificado.get(key)) mapaUnificado.set(key, intensidad);
315 } else {
316 mapaUnificado.set(key, intensidad);
317 }
318 });
319 });
320
321 const puntosFinales = [];
322 const arrayParaCache = [];
323
324 mapaUnificado.forEach((intensidad, key) => {
325 const [lat, lng] = key.split('_').map(parseFloat);
326 puntosFinales.push([lat, lng, intensidad]);
327
328 // Convertimos Intensidad a valor numérico ICA legible
329 let valorICA = 0;
330 if (intensidad >= 0.2) valorICA = 25;
331 if (intensidad >= 0.6) valorICA = 75;
332 if (intensidad >= 1.0) valorICA = 150;
333 arrayParaCache.push([lat, lng, valorICA]);
334 });
335
336 globalDatosCache = arrayParaCache;
337
338 return {
339 puntos: puntosFinales,
340 gradiente: { 0.0: 'rgba(0,0,0,0)', 0.2: 'green', 0.6: 'yellow', 1.0: 'red' },
341 unidad: "ICA Global"
342 };
343}
