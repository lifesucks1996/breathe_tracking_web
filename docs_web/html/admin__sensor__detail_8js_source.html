112/**
113 * @brief Carga los datos del sensor basándose en la URL.
114 * @details
115 * 1. Lee el parámetro `?id=` de la URL.
116 * 2. Busca ese ID en `adminSensorsData` (cargado desde admin_sensors_data.js).
117 * 3. Actualiza el objeto `window.sensorDetailData` con la información real.
118 * 4. Actualiza el DOM (Título, Batería, Ubicación, Última Conexión).
119 */
120function loadSensorData() {
121 // 1. LEER EL ID DE LA URL
122 const params = new URLSearchParams(window.location.search);
123 const sensorId = params.get('id');
124
125 // 2. BUSCAR EL SENSOR EN LA LISTA GLOBAL
126 // Por defecto usamos el mock que ya existía
127 let sensor = window.sensorDetailData;
128
129 // Si tenemos la lista de sensores cargada y hay un ID en la URL, buscamos el real
130 if (sensorId && typeof adminSensorsData !== 'undefined') {
131 const sensorEncontrado = adminSensorsData.find(s => s.id === sensorId);
132
133 if (sensorEncontrado) {
134 // Construimos el objeto de detalle con los datos del sensor encontrado
135 sensor = {
136 ...window.sensorDetailData, // Mantenemos datos dummy (batería, path) si faltan
137 id: sensorEncontrado.id,
138 name: sensorEncontrado.name || sensorEncontrado.nombre,
139 currentLocation: sensorEncontrado.location || sensorEncontrado.ubicacion,
140 lastConnection: sensorEncontrado.lastConnection,
141 pathCoords: [sensorEncontrado.coords] // Centrar mapa en este sensor
142 };
143 // Actualizamos la variable global para que initDetailMap la use
144 window.sensorDetailData = sensor;
145 }
146 }
147
148 const data = sensor;
149
150 // --- RENDERIZADO EN EL DOM ---
151
152 // Título
153 document.getElementById('sensor-detail-title').textContent = data.name;
154
155 // Última conexión
156 const lastConnInfo = document.getElementById('last-connection-info');
157 if(lastConnInfo) {
158 lastConnInfo.innerHTML = `Última conex: <span style="color:#e74c3c">${data.lastConnection}</span>`;
159 }
160
161 // Batería
162 const battElem = document.getElementById('battery-status-text');
163 if(battElem) {
164 battElem.textContent = `Batería ${data.battery}`;
165 if(parseInt(data.battery) < 20) {
166 battElem.innerHTML += ' <i class="fas fa-battery-quarter" style="color:red"></i>';
167 } else {
168 battElem.innerHTML += ' <i class="fas fa-battery-three-quarters" style="color:green"></i>';
169 }
170 }
171
172 // Ubicación
173 const curLoc = document.getElementById('current-location');
174 if(curLoc) curLoc.textContent = data.currentLocation;
175
176 // Ocultar segunda línea de ubicación para limpieza visual
177 const avgLoc = document.getElementById('avg-location');
178 if(avgLoc) {
179 avgLoc.style.display = 'none';
180 }
181
182 // Datos del punto inferior (Métricas actuales)
183 const pLoc = document.getElementById('point-location-text');
184 if(pLoc && data.point4) pLoc.textContent = data.currentLocation;
185
186 const pTime = document.getElementById('point-time');
187 if(pTime) pTime.textContent = `Hora: ${data.lastConnection}`;
188}

194/**
195 * @brief Mapea el nombre del contaminante en la UI al ID de la base de datos.
196 * @param {string} uiName - Nombre seleccionado en el dropdown (ej: "Ozono").
197 * @return {string} ID del documento en Firestore (ej: "ozono").
198 */
199function mapContaminantToId(uiName) {
200 switch (uiName) {
201 case 'Ozono': case 'ozono': return 'ozono';
202 case 'CO2': case 'co2': return 'co2';
203 case 'CO': case 'co': return 'co';
204 case 'NO2': case 'no2': return 'no2';
205 case 'SO2': case 'so2': return 'so2';
206 default: return '';
207 }
208}

210/**
211 * @brief Renderiza o actualiza el gráfico de actividad.
212 * @details
213 * Obtiene datos históricos de Firestore (colección "datos_grafico")
214 * y los pinta usando Chart.js.
215 * * @param {string} contaminantType - Tipo de contaminante a visualizar.
216 * @async
217 */
218async function renderChart(contaminantType) {
219 const ctx = document.getElementById('activityChart')?.getContext('2d');
220 if (!ctx) return;
221
222 let dataValues = [];
223 let unit = '';
224 let labelText = `Nivel de ${contaminantType}`;
225 let borderColor = '#95a5a6';
226
227 const queryType = mapContaminantToId(contaminantType);
228
229 // Configuración de colores y unidades según el gas
230 switch (queryType) {
231 case 'ozono': borderColor = '#8A2BE2'; unit = 'ppm'; break;
232 case 'co2': borderColor = '#32CD32'; unit = 'ppm'; break;
233 case 'co': borderColor = '#FF8C00'; unit = 'mg/m³'; break;
234 case 'no2': borderColor = '#E6A100'; unit = 'µg/m³'; break;
235 case 'so2': borderColor = '#00BFFF'; unit = 'µg/m³'; break;
236 default: borderColor = '#95a5a6';
237 }
238
239 let bgColor = borderColor.replace('rgb', 'rgba').replace(')', ', 0.2)');
240 if (bgColor.startsWith('#')) bgColor = 'rgba(100, 100, 100, 0.2)';
241
242 // Obtención de datos desde Firestore
243 if (db && queryType) {
244 try {
245 const docRef = doc(db, "datos_grafico", queryType);
246 const docSnap = await getDoc(docRef);
247
248 if (docSnap.exists()) {
249 const data = docSnap.data();
250 dataValues = data.valor || [];
251 }
252 } catch (e) {
253 console.error("Error FB:", e);
254 }
255 }
256
257 // Relleno de seguridad si no hay datos
258 if (dataValues.length !== HOURLY_LABELS.length) {
259 dataValues = HOURLY_LABELS.map(() => null);
260 }
261
262 // Destruir gráfico previo si existe para evitar superposiciones
263 if (sensorChart) sensorChart.destroy();
264
265 // Crear nuevo gráfico
266 sensorChart = new Chart(ctx, {
267 type: 'line',
268 data: {
269 labels: HOURLY_LABELS,
270 datasets: [{
271 label: labelText,
272 data: dataValues,
273 borderColor: borderColor,
274 backgroundColor: bgColor,
275 borderWidth: 2,
276 tension: 0.4,
277 fill: false,
278 pointRadius: 3
279 }]
280 },
281 options: {
282 responsive: true,
283 maintainAspectRatio: false,
284 plugins: { legend: { display: false } },
285 interaction: { mode: 'nearest', axis: 'x', intersect: false },
286 scales: {
287 y: { beginAtZero: true, title: { display: true, text: unit } },
288 x: { grid: { display: false } }
289 }
290 }
291 });
292}
