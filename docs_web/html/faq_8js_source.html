23/**
24 * @brief Inicializa todos los elementos FAQ y asigna listeners de eventos.
25 * @details
26 * Selecciona todos los elementos con clase 'faq-question' y añade un

41/**
42 * @brief Alterna la apertura/cierre de una respuesta FAQ con animación suave.
43 * @details
44 * Si la pregunta está cerrada, la abre y hace scroll suave hacia ella.
45 * Si está abierta, simplemente la cierra.
46 * Actualiza el atributo aria-expanded para accesibilidad.
47 *
48 * @param {Element} elemento - El botón .faq-question clickeado
49 * @return void
50 */
51function abrirCerrarRespuesta(elemento) {
52 const item = elemento.closest('.faq-item');
53 const yaAbierto = item.classList.contains('active');
54
55 // Cerrar todos los items abiertos (opcional: descomenta para acordeón)
56 // cerrarTodas();
57
58 // Alternar el estado del item actual
59 if (yaAbierto) {
60 item.classList.remove('active');
61 // Animar el cierre
62 elemento.setAttribute('aria-expanded', 'false');
63 } else {
64 item.classList.add('active');
65 // Animar la apertura
66 elemento.setAttribute('aria-expanded', 'true');
67
68 // Scroll suave hacia la pregunta abierta
69 elemento.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
70 }
71}

92/**
93 * @brief Inicializa la navegación desde la tabla de contenidos con scroll suave.
94 * @details
95 * Asigna listeners a todos los enlaces de la tabla de contenidos.
96 * Cuando se hace clic, realiza un scroll suave hacia la sección destino
97 * y abre automáticamente el primer FAQ de esa sección.
98 *
99 * void -> void
100 * @return void
101 */
102function inicializarScrollNavegacion() {
103 const enlaces = document.querySelectorAll('.table-of-contents a');
104
105 enlaces.forEach(enlace => {
106 enlace.addEventListener('click', function(e) {
107 e.preventDefault();
108
109 const idDestino = this.getAttribute('href').substring(1);
110 const elemento = document.getElementById(idDestino);
111
112 if (elemento) {
113 // Scroll suave hacia la sección
114 elemento.scrollIntoView({ behavior: 'smooth' });
115
116 // Opcional: Abrir el primer FAQ de la sección
117 abrirPrimerFAQDelaSección(elemento);
118 }
119 });
120 });
121}

123/**
124 * @brief Abre el primer item FAQ de una sección específica.
125 * @details
126 * Busca el primer elemento .faq-item dentro de la sección proporcionada.
127 * Si existe y no está activo, lo activa y actualiza aria-expanded.
128 *
129 * @param {Element} seccion - La sección (elemento DOM) a procesar
130 * @return void
131 */
132function abrirPrimerFAQDelaSección(seccion) {
133 const primerItem = seccion.querySelector('.faq-item');
134 if (primerItem && !primerItem.classList.contains('active')) {
135 primerItem.classList.add('active');
136 const boton = primerItem.querySelector('.faq-question');
137 boton.setAttribute('aria-expanded', 'true');
138 }
139}

141/**
142 * @brief Busca preguntas FAQ según un término de búsqueda.
143 * @details
144 * Filtra los items FAQ mostrando solo aquellos cuya pregunta o respuesta
145 * contenga el término buscado (case-insensitive).
146 * Oculta el resto mediante display: 'none'.
147 *
148 * @param {string} termino - El término a buscar en preguntas y respuestas
149 * @return void
150 */
151function buscarPreguntas(termino) {
152 const items = document.querySelectorAll('.faq-item');
153 const terminoLower = termino.toLowerCase();
154
155 items.forEach(item => {
156 const pregunta = item.querySelector('.faq-question .question-text').textContent.toLowerCase();
157 const respuesta = item.querySelector('.faq-answer').textContent.toLowerCase();
158
159 if (pregunta.includes(terminoLower) || respuesta.includes(terminoLower)) {
160 item.style.display = 'block';
161 } else {
162 item.style.display = 'none';
163 }
164 });
165}

167/**
168 * @brief Listener de teclado para navegar FAQ con Tab + Enter (accesibilidad).
169 * @details
170 * Permite a usuarios navegar el FAQ usando solo teclado.
171 * Enter abre/cierra la pregunta enfocada.
172 * Previene el comportamiento por defecto de Enter cuando se enfoca un .faq-question.
173 *
174 * @event keydown - Se dispara cuando se presiona una tecla
175 * @return void
176 */
177document.addEventListener('keydown', function(event) {
178 if (event.key === 'Enter') {
179 const elemento = document.activeElement;
180 if (elemento.classList.contains('faq-question')) {
181 event.preventDefault();
182 abrirCerrarRespuesta(elemento);
183 }
184 }
185});
