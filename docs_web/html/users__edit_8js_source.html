74/**
75 * @brief Valida los datos del formulario aplicando reglas de negocio.

77function validate(values) { // <--- HE QUITADO EL 'export'
78 let ok = true;
79
80 // Validación de Nombre
81 if (!values.nombre || values.nombre.trim().length < 2) {
82 showFieldError('nombre', 'El nombre es obligatorio (mínimo 2 caracteres).');
83 ok = false;
84 }
85
86 // Validación de Apellidos
87 if (!values.apellidos || values.apellidos.trim().length < 2) {
88 showFieldError('apellidos', 'Los apellidos son obligatorios.');
89 ok = false;
90 }
91
92 // Validación de Código Postal (Estricto: exactamente 5 dígitos numéricos)
93 if (values.cp && values.cp.trim().length > 0 && !/^\d{5}$/.test(values.cp)) {
94 showFieldError('cp', 'El Código Postal debe ser de 5 dígitos.');
95 ok = false;
96 }
97
98 // Validación de cambio de contraseña
99 const hasPasswordFields = values.currentPassword || values.newPassword || values.confirmPassword;
100
101 if (hasPasswordFields) {
102 if (!values.currentPassword || values.currentPassword.trim().length === 0) {
103 showFieldError('currentPassword', 'La contraseña actual es obligatoria para cambiar la contraseña.');
104 ok = false;
105 }
106
107 if (!values.newPassword || values.newPassword.trim().length === 0) {
108 showFieldError('newPassword', 'La nueva contraseña es obligatoria.');
109 ok = false;
110 } else {
111 const passValidation = validatePasswordStrength(values.newPassword);
112 if (!passValidation.valid) {
113 showFieldError('newPassword', passValidation.message);
114 ok = false;
115 }
116 }
117
118 if (!values.confirmPassword || values.confirmPassword.trim().length === 0) {
119 showFieldError('confirmPassword', 'Debes confirmar la nueva contraseña.');
120 ok = false;
121 } else if (values.newPassword !== values.confirmPassword) {
122 showFieldError('confirmPassword', 'Las contraseñas no coinciden.');
123 ok = false;
124 }
125 }
126
127 return ok;
128}

130/**
131 * @brief Valida la complejidad de una contraseña según los requisitos.
132 */
133function validatePasswordStrength(password) { // <--- HE QUITADO EL 'export'
134 const requirements = {
135 length: password.length >= 8,
136 uppercase: /[A-Z]/.test(password),
137 number: /\d/.test(password),
138 special: /[\W_]/.test(password)
139 };
140
141 const allValid = Object.values(requirements).every(req => req);
142
143 if (!allValid) {
144 let missing = [];
145 if (!requirements.length) missing.push('8 caracteres');
146 if (!requirements.uppercase) missing.push('una mayúscula');
147 if (!requirements.number) missing.push('un número');
148 if (!requirements.special) missing.push('un carácter especial');
149
150 return {
151 valid: false,
152 message: `La contraseña debe contener: ${missing.join(', ')}.`,
153 requirements
154 };
155 }
156
157 return { valid: true, requirements };
158}
