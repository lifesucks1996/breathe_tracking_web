47def generar_clusters(lat_centro, lng_centro, num_clusters=3):
48 """
49 @brief Genera centros de clusters de contaminación
50 @param lat_centro Centro base latitud
51 @param lng_centro Centro base longitud
52 @param num_clusters Número de focos de contaminación
53 @return Lista de tuplas (lat, lng) con los centros
54 """
55 clusters = []
56 for _ in range(num_clusters):
57 # Desplazamiento del cluster respecto al centro
58 offset_lat = random.uniform(-RADIO_VARIACION, RADIO_VARIACION)
59 offset_lng = random.uniform(-RADIO_VARIACION, RADIO_VARIACION)
60 clusters.append((lat_centro + offset_lat, lng_centro + offset_lng))
61 return clusters

63def generar_puntos_alrededor_cluster(lat_cluster, lng_cluster, num_puntos, radio_cluster):
64 """
65 @brief Genera puntos concentrados alrededor de un cluster
66 @param lat_cluster Centro del cluster
67 @param lng_cluster Centro del cluster
68 @param num_puntos Número de puntos a generar
69 @param radio_cluster Radio de dispersión del cluster
70 @return Lista de tuplas (lat, lng)
71 """
72 puntos = []
73 for _ in range(num_puntos):
74 # Distribución gaussiana para concentrar puntos en el centro
75 angulo = random.uniform(0, 2 * math.pi)
76 # Usar distribución normal para que los puntos estén más cerca del centro
77 distancia = abs(random.gauss(0, radio_cluster / 2))
78 distancia = min(distancia, radio_cluster) # Limitar al radio máximo
79
80 lat = lat_cluster + distancia * math.cos(angulo)
81 lng = lng_cluster + distancia * math.sin(angulo)
82 puntos.append((lat, lng))
83 return puntos
