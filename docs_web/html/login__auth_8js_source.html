38/**
39 * @brief Recupera el perfil extendido del usuario desde Firestore para determinar su rol.
40 * userId: string -> fetchUserProfile() -> Promise<Object>
41 *
42 * @details
43 * Accede a la colección "Usuarios" utilizando el UID proporcionado por Authentication.
44 * Extrae el campo 'Id_rol' para normalizarlo y devolverlo junto con el resto de datos.
45 * Si el documento no existe, asigna un rol por defecto ('usuario') para evitar bloqueos.
46 *
47 * @note Es crucial para la seguridad de la navegación, ya que el Auth de Firebase
48 * por sí solo no almacena roles personalizados en este esquema.
49 * * @param userId El identificador único (UID) del usuario autenticado.
50 * @return Promesa que resuelve en un Objeto con los datos del usuario y la propiedad 'rol'.
51 */
52async function fetchUserProfile(userId) {
53 const docRef = doc(db, "Usuarios", userId);
54 const docSnap = await getDoc(docRef);
55
56 if (docSnap.exists()) {
57 const userData = docSnap.data();
58 // Mapeamos Id_rol a la variable 'rol' para la lógica de redirección
59 return {
60 ...userData,
61 rol: userData.Id_rol || 'usuario'
62 };
63 }
64 // Fallback seguro en caso de inconsistencia en la base de datos
65 return { rol: 'usuario' };
66}
