29/**
30 * @brief Inicializa el sistema de estaciones oficiales en el mapa proporcionado.
31 * @details
32 * 1. Crea la capa `capaEstaciones` y la añade al mapa.
33 * 2. Carga las estaciones visibles inicialmente.
34 * 3. Configura un listener `moveend` para recargar estaciones al mover el mapa (con debounce de 1s).
35 *
36 * @param {L.Map} mapa - Instancia del mapa Leaflet donde se pintarán las estaciones.
37 */
38function inicializarEstacionesOficiales(mapa) {
39 if (!mapa) return; // Validación de seguridad
40
41 console.log(" Iniciando sistema de estaciones oficiales...");
42
43 // Crear grupo de capas para gestión eficiente
44 capaEstaciones = L.layerGroup().addTo(mapa);
45
46 // Carga inicial
47 cargarEstaciones(mapa);
48
49 // Configuración del listener con debounce (retraso)
50 let timeout;
51 mapa.on('moveend', () => {
52 // Limpia el timeout anterior para evitar peticiones múltiples
53 clearTimeout(timeout);
54 // Espera 1 segundo de inactividad antes de recargar
55 timeout = setTimeout(() => cargarEstaciones(mapa), 1000);
56 });
57}

59/**
60 * @brief Consulta la API de WAQI para obtener estaciones dentro de los límites visibles.
61 * @details
62 * Obtiene los bounds (Norte, Sur, Este, Oeste) del mapa y construye la URL de consulta.
63 * Si la respuesta es exitosa, llama a `dibujarEstacionesComoIconos`.
64 *
65 * @param {L.Map} mapa - Instancia del mapa para obtener los límites (bounds).
66 * @async
67 */
68async function cargarEstaciones(mapa) {
69 // Obtener límites visibles
70 const bounds = mapa.getBounds();
71 const lat1 = bounds.getSouth();
72 const lng1 = bounds.getWest();
73 const lat2 = bounds.getNorth();
74 const lng2 = bounds.getEast();
75
76 console.log(` Consultando estaciones...`);
77
78 // Construcción de URL geo-restringida
79 const url = `https://api.waqi.info/map/bounds/?latlng=${lat1},${lng1},${lat2},${lng2}&token=${WAQI_TOKEN}`;
80
81 try {
82 const respuesta = await fetch(url);
83 const datos = await respuesta.json();
84
85 if (datos.status === 'ok') {
86 console.log(` Estaciones encontradas: ${datos.data.length}`);
87 dibujarEstacionesComoIconos(datos.data);
88 } else {
89 console.warn(" API WAQI Error:", datos.data);
90 }
91 } catch (error) {
92 console.error(" Error conectando con estaciones:", error);
93 }
94}

96/**
97 * @brief Renderiza las estaciones recibidas como iconos en el mapa.
98 * @details
99 * 1. Limpia los marcadores antiguos de `capaEstaciones`.
100 * 2. Itera sobre cada estación recibida.
101 * 3. Crea un `L.divIcon` personalizado con el color correspondiente al AQI.
102 * 4. Añade un tooltip y un evento click para ver detalles.
103 *
104 * @param {Array<Object>} estaciones - Lista de estaciones devuelta por la API.
105 */
106function dibujarEstacionesComoIconos(estaciones) {
107 // Limpieza previa
108 capaEstaciones.clearLayers();
109
110 estaciones.forEach(est => {
111 // Filtrar estaciones sin datos validos
112 if (est.aqi === '-') return;
113
114 const valorAQI = parseInt(est.aqi);
115 const colorFondo = obtenerColorAQI(valorAQI);
116
117 // Icono HTML personalizado
118 const iconoOficial = L.divIcon({
119 className: '', // Sin clases por defecto de Leaflet
120 html: `
121 <div class="station-icon-marker" style="background-color: ${colorFondo}; width: 30px; height: 30px;">
122 <i class="fas fa-broadcast-tower"></i>
123 </div>
124 `,
125 iconSize: [30, 30],
126 iconAnchor: [15, 15],
127 popupAnchor: [0, -20]
128 });
129
130 // Creación del marcador
131 const marker = L.marker([est.lat, est.lon], {
132 icon: iconoOficial,
133 zIndexOffset: 1000 // Prioridad visual alta
134 });
135
136 // Tooltip básico (hover)
137 marker.bindTooltip(`
138 <div style="text-align:center">
139 <strong>${est.station.name}</strong><br>
140 Estación Oficial<br>
141 AQI: ${est.aqi}
142 </div>
143 `, { direction: 'top', offset: [0, -20] });
144
145 // Evento Click: Cargar detalles
146 marker.on('click', (e) => {
147 L.DomEvent.stopPropagation(e); // Evitar propagación al mapa
148 mostrarPopupDetalle(est, marker);
149 });
150
151 capaEstaciones.addLayer(marker);
152 });
153}

155/**
156 * @brief Carga y muestra los detalles completos de una estación en un popup.
157 * @details
158 * Realiza una segunda petición a la API usando el UID de la estación para obtener
159 * desglose de contaminantes (PM2.5, PM10, O3, etc.) y metadatos.
160 *
161 * @param {Object} estacionResumen - Datos básicos de la estación (incluye UID).
162 * @param {L.Marker} marker - El marcador sobre el cual abrir el popup.
163 * @async
164 */
165async function mostrarPopupDetalle(estacionResumen, marker) {
166 const popupOptions = {
167 autoPan: false, // Evita mover el mapa para no disparar recargas
168 className: 'official-popup-container'
169 };
170
171 // Popup temporal de carga
172 marker.bindPopup('<div style="padding:10px; text-align:center; color:#5BA3F5"><i class="fas fa-spinner fa-spin"></i> Conectando...</div>', popupOptions).openPopup();
173
174 // URL de detalle por UID
175 const url = `https://api.waqi.info/feed/@${estacionResumen.uid}/?token=${WAQI_TOKEN}`;
176
177 try {
178 const res = await fetch(url);
179 const json = await res.json();
180
181 if (json.status === 'ok') {
182 const d = json.data;
183 const iaqa = d.iaqi;
184
185 // Diccionario de traducción de contaminantes
186 const nombres = {
187 pm25: "PM 2.5", pm10: "PM 10", o3: "Ozono (O₃)",
188 no2: "NO₂", so2: "SO₂", co: "CO",
189 t: "Temp.", h: "Humedad", w: "Viento", p: "Presión"
190 };
191
192 // Generación de lista HTML
193 let listadoHtml = '';
194 for (const [key, val] of Object.entries(iaqa)) {
195 if (nombres[key]) {
196 listadoHtml += `
197 <li>
198 <span class="gas-name">${nombres[key]}</span>
199 <span class="gas-value">${val.v}</span>
200 </li>`;
201 }
202 }
203
204 // Construcción del HTML final
205 const html = `
206 <div class="official-popup">
207 <div class="popup-header" style="background-color: ${obtenerColorAQI(d.aqi)}">
208 <span class="city-name">${d.city.name.split(',')[0]}</span>
209 <span class="aqi-badge">AQI ${d.aqi}</span>
210 </div>
211 <div class="popup-body">
212 <div class="last-update">
213 <i class="fas fa-clock"></i> Hora: ${d.time.s.substring(11, 16)}
214 </div>
215 <ul class="pollutant-list">
216 ${listadoHtml || '<li>Datos generales solo</li>'}
217 </ul>
218 <div class="popup-footer">
219 Fuente: ${d.attributions[0]?.name || 'GVA'}
220 </div>
221 </div>
222 </div>
223 `;
224 marker.setPopupContent(html);
225 }
226 } catch (e) {
227 marker.setPopupContent('Error de conexión.');
228 }
229}

231/**
232 * @brief Devuelve un color hexadecimal según el nivel de AQI.
233 * @details
234 * - 0-50: Verde (Bueno)
235 * - 51-100: Amarillo (Moderado)
236 * - 101-150: Naranja (Insalubre para grupos sensibles)
237 * - 151-200: Rojo (Insalubre)
238 * - 201-300: Morado (Muy insalubre)
239 * - >300: Granate (Peligroso)
240 *
241 * @param {number} aqi - Índice de Calidad del Aire.
242 * @return {string} Color en formato Hex.
243 */
244function obtenerColorAQI(aqi) {
245 if (aqi <= 50) return "#009966";
246 if (aqi <= 100) return "#ffde33";
247 if (aqi <= 150) return "#ff9933";
248 if (aqi <= 200) return "#cc0033";
249 if (aqi <= 300) return "#660099";
250 return "#7e0023";
251}
