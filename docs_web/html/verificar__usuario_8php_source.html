54/**
55 * @brief Decodifica una cadena Base64 URL-Safe recibida por GET.
56 * input:string -> decodificar_parametro() -> string
57 * * @details
58 * Reinvierte el proceso de codificación realizado en Python/JS.
59 * 1. Restaura el padding '=' si fue truncado.
60 * 2. Reemplaza caracteres URL-safe (-_) por los estándar de Base64 (+/).
61 * 3. Decodifica la cadena resultante.
62 * * @param input La cadena codificada recibida en la URL.
63 * @return La cadena original decodificada (email o token).
64 */
65function decodificar_parametro($input) {
66 $remainder = strlen($input) % 4;
67 if ($remainder) {
68 $padlen = 4 - $remainder;
69 $input .= str_repeat('=', $padlen);
70 }
71 return base64_decode(strtr($input, '-_', '+/'));
72}
